code: |
  import pandas as pd
  from mlxtend.preprocessing import TransactionEncoder
  from mlxtend.frequent_patterns import apriori, association_rules

  baskets = [['bread', 'milk'], ['bread', 'milk'], ['bread'], ['tea']]
  encoder = TransactionEncoder()
  encoded = encoder.fit(baskets).transform(baskets)
  table = pd.DataFrame(encoded, columns=encoder.columns_)
  groups = apriori(table, min_support=0.5, use_colnames=True)
  rules = association_rules(groups, metric='confidence', min_threshold=0.6)
  print(rules[['antecedents', 'consequents', 'support', 'confidence', 'lift']])

تخيّل أربع سلال. ثلاث منها تحتوي على خبز، واثنتان من هذه الثلاث تحتويان أيضًا على حليب. في قاعدة **الخبز ← الحليب**، <mark>الدعم</mark> هو نسبة السلال التي تضمّ العنصرين من جميع السلال: 2 من 4. و<mark>الثقة</mark> هي نسبة سلال الخبز التي تضمّ حليبًا: 2 من 3.

لحساب <mark>الرفع</mark>، اقسم الثقة على نسبة السلال التي تحتوي على الحليب من جميع السلال. هذه النسبة هنا هي النصف، فالرفع يساوي نحو 1.33. إذا تجاوز الرفع 1، فالحليب أكثر شيوعًا في سلال الخبز منه في السلال عمومًا. وإذا كان أقل من 1، فالحليب أقل شيوعًا فيها.

تختصر أبريوري العمل بملاحظة بسيطة: كل سلّة فيها خبز وحليب وبيض فيها بالضرورة خبز وحليب. إذا كان ظهور الزوج نادرًا، فلن يبلغ الثلاثي الحد نفسه للدعم.
