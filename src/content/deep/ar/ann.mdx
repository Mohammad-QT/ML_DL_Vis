  from sklearn.datasets import make_circles
  from sklearn.model_selection import train_test_split
  from sklearn.neural_network import MLPClassifier
  from sklearn.pipeline import Pipeline
  from sklearn.preprocessing import StandardScaler

  X, y = make_circles(n_samples=400, factor=0.35, noise=0.08, random_state=42)
  X_train, X_test, y_train, y_test = train_test_split(
      X, y, test_size=0.25, stratify=y, random_state=42
  )
  model = Pipeline([
      ("scale", StandardScaler()),
      ("mlp", MLPClassifier(hidden_layer_sizes=(8, 8), activation="relu", max_iter=1000, random_state=42)),
  ])
  model.fit(X_train, y_train)
  print(model.score(X_test, y_test))

يشبه <mark>العصبون</mark> الواحد نموذج الانحدار اللوجستي: يضرب كل مدخل في وزنه، ويضيف قيمة انحياز، ثم يمرّر الناتج عبر دالة تفعيل. عندما تجمع عدّة وحدات في طبقة مخفية وتمرّر مخرجاتها إلى الطبقة التالية، تنتهي بوحدة مخرج تحوّل الدرجة إلى احتمال الفئة ب. في مخطّط الشبكة تدلّ الخطوط الزرقاء على الأوزان الموجبة، والوردية على السالبة، وكلّما زاد سُمك الخط كان الوزن أكبر.

تتيح <mark>دالة التفعيل</mark> (مثل <bdi>ReLU</bdi> أو <bdi>tanh</bdi>) للحدّ الفاصل أن ينحني. من دون هذا العنصر غير الخطّي، سيبقى تكديس الطبقات مكافئًا لخط مستقيم واحد. أمّا بوجوده، فتدمج الوحدات المخفية خطوطًا بسيطة لتصنع منحنيات تلتفّ حول الحلقات أو تتبع شكلًا لولبيًّا.

تضغط دالة <mark><bdi>Sigmoid</bdi></mark> أي رقم بين 0 و1، ولذلك تستخدمها وحدة المخرج لإعطاء احتمال الفئة ب، بينما تعمّم <bdi>softmax</bdi> الفكرة نفسها على عدّة فئات كما في درس <bdi>CNN</bdi>. وفي الطبقات المخفية تضغط <bdi>tanh</bdi> القيم بين <bdi>−1</bdi> و1 وتتمركز حول الصفر. أمّا <mark><bdi>ReLU</bdi></mark> فتُبقي القيم الموجبة وتحوّل السالبة إلى 0، ما يسرّع التدريب لكنّه قد يُبقي بعض الوحدات عالقة عند 0 («تموت»). وتحتفظ <bdi>Leaky ReLU</bdi> بميل صغير (0.1) للقيم السالبة كي لا تموت الوحدات تمامًا.

يتكرّر التدريب في حلقة بسيطة على دفعات صغيرة من النقاط. يُنتج التمرير الأمامي توقّعات ويقيس <mark>خسارة الإنتروبيا المتقاطعة</mark>. بعد ذلك يرجع <mark>الانتشار الخلفي</mark> لحساب إسهام كل وزن في الخطأ، ويحرّك الانحدار التدرّجي كل وزن خطوة صغيرة لتقليل الخسارة. يُحتفَظ بنحو ربع النقاط محاطة بحلقات كمجموعة اختبار لتتأكّد من أن الشبكة تعمّم جيّدًا.
