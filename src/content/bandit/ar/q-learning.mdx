  import numpy as np
  from sklearn.metrics import mean_absolute_error

  grid = ["......", ".##.X.", "...#..", ".X.#..", "..#...", "S....G"]
  moves = [(-1, 0), (0, 1), (1, 0), (0, -1)]
  rng = np.random.default_rng(2026)
  q = np.zeros((6, 6, 4))
  steps, goal_steps = [], []

  for episode in range(400):
      row, col = 5, 0
      for count in range(1, 73):
          values = q[row, col]
          best = np.flatnonzero(values == values.max())
          action = rng.integers(4) if rng.random() < .2 else rng.choice(best)
          dr, dc = moves[action]
          nr, nc = row + dr, col + dc
          if not (0 <= nr < 6 and 0 <= nc < 6) or grid[nr][nc] == "#":
              nr, nc = row, col
          tile = grid[nr][nc]
          reward = {"G": 1, "X": -1}.get(tile, -.04)
          terminal = tile in "GX"
          future = 0 if terminal else q[nr, nc].max()
          q[row, col, action] += .5 * (reward + .95 * future - values[action])
          row, col = nr, nc
          if terminal:
              break
      steps.append(count)
      if tile == "G":
          goal_steps.append(count)

  recent = goal_steps[-30:]
  print(mean_absolute_error([5] * len(recent), recent))

ينتهي اختيار ذراع في مسألة اللصوص متعدّدي الأذرع بمكافأة. أمّا التحرّك في الشبكة فيغيّر أيضًا موقع الوكيل في الخطوة التالية. وقد يحتاج إلى عدة خطوات ليدور حول جدار قبل أن يحصل على مكافأة الهدف.

يحتفظ تعلّم Q بـ<mark>درجة لكل حركة في كل خلية</mark>. يحدّث درجة واحدة في كل مرة، فيقرّبها من مجموع المكافأة التي تلقّاها وأفضل درجة متوقعة في الخلية التالية بعد خصمها. يتحكّم معدل التعلّم في مقدار التغيير، ويتحكّم الخصم في وزن المكافآت المستقبلية. تنتهي الحلقة عند بلوغ الهدف أو السقوط في حفرة، فلا تبقى درجة مستقبلية لإضافتها.
