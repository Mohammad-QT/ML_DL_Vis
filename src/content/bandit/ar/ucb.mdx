code: |
  import numpy as np
  url = "https://raw.githubusercontent.com/Mohammad-QT/ML_DL_Vis/main/datasets/ad_clicks.csv"
  clicks = np.loadtxt(url, delimiter=",", skiprows=1, dtype=int)
  plays = np.zeros(clicks.shape[1], dtype=int)
  wins = np.zeros_like(plays)
  for t in range(len(clicks)):
      if t < len(plays):
          ad = t
      else:
          scores = wins / plays + np.sqrt(2 * np.log(t) / plays)
          ad = np.argmax(scores)
      reward = clicks[t, ad]
      plays[ad] += 1
      wins[ad] += reward
  print(wins.sum())

لديك خمسة إعلانات ولا يمكنك عرض سوى إعلان واحد في كل جولة. لكل إعلان احتمال نقر خفيّ. قيمة النقرة 1 وقيمة عدم النقر 0. لا تعرف سوى نتيجة الإعلان الذي عرضته. تُسمّى هذه <mark>مسألة قاطع الطريق متعدد الأذرع</mark>: تخيّل خمس آلات لعب، لكن لا يمكنك سحب سوى ذراع واحدة في كل جولة.

هل تواصل عرض أفضل إعلان حتى الآن، أم تجرّب إعلانًا لا تعرف عنه الكثير؟ الخيار الأول هو <mark>الاستغلال</mark>، والثاني هو <mark>الاستكشاف</mark>. توازن الخوارزمية بينهما بإعطاء الإعلانات الأقل تجربة مكافأة أكبر. وتظل النقرة خاضعة للصدفة؛ فقد لا يحصل أفضل إعلان على نقرة، والسياسة التي تعرفه مسبقًا مرجع للمقارنة وليست ضمانًا لأكبر عدد من النقرات في كل تشغيل.
