code: |
  import numpy as np
  url = "https://raw.githubusercontent.com/Mohammad-QT/ML_DL_Vis/main/datasets/ad_clicks.csv"
  clicks = np.loadtxt(url, delimiter=",", skiprows=1, dtype=int)
  plays = np.zeros(clicks.shape[1], dtype=int)
  wins = np.zeros_like(plays)
  rng = np.random.default_rng(2026)
  for t in range(len(clicks)):
      samples = rng.beta(1 + wins, 1 + plays - wins)
      ad = np.argmax(samples)
      reward = clicks[t, ad]
      plays[ad] += 1
      wins[ad] += reward
  print(wins.sum())

تخيّل أنك تجرّب إعلانًا جديدًا. بعد نقرة واحدة، قد يظل معدله الحقيقي منخفضًا أو مرتفعًا. بدلًا من إعطاء الإعلان درجة ثابتة، تحتفظ طريقة تومبسون بـ<mark>توزيع</mark>: منحنى يصف معدلات النقر المعقولة بناءً على النتائج حتى الآن.

في كل جولة، اسحب معدلًا محتملًا من كل منحنى واعرض الإعلان صاحب أعلى عينة. قد يعطيك إعلان لم تجرّبه كثيرًا معدلًا عاليًا فيحصل على فرصة أخرى. ومع تراكم النتائج، تميل الإعلانات الجيدة إلى إعطاء معدلات عالية أكثر من غيرها. وكما في طريقة الحد الأعلى للثقة، تتعلّم هذه السياسة من نتيجة الإعلان الذي تعرضه فقط.
