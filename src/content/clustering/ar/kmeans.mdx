code: |
  import pandas as pd
  from sklearn.cluster import KMeans

  url = "https://raw.githubusercontent.com/Mohammad-QT/ML_DL_Vis/main/datasets/clustering_points.csv"
  X = pd.read_csv(url)[["x", "y"]]
  model = KMeans(n_clusters=3, init="k-means++", n_init=10, random_state=42)
  groups = model.fit_predict(X)
  print(model.cluster_centers_, model.inertia_)

حتى لو لم تكن لنقاطك تسميات، يمكنك أن تسأل: أيّ النقاط تنتمي إلى المجموعة نفسها؟ ترسم الخوارزمية مجموعات حول <mark>مراكز</mark>. وهي لا تعرف ما تعنيه كل مجموعة؛ بل تعتمد فقط على المسافات بين النقاط والمراكز.

يعرض مخطّط الكوع **مجموع مربّعات المسافات داخل المجموعات** (<bdi>WCSS</bdi>): ربّع مسافة كل نقطة إلى أقرب مركز ثم اجمع المربّعات. يجرّب المخطّط عدّة بدايات لكل عدد من المراكز ويعرض أقلّ قيمة يجدها. استخدم الانحناء مؤشرًا، لا برهانًا على صحّة اختيارك.
